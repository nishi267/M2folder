# ══════════════════════════════════════════════════════════════
# SCORING ENGINE — no changes needed below this line
# ══════════════════════════════════════════════════════════════

def apply_bins(value, bins, inverse):
    """
    Apply bin rules to a single value.

    Normal  (inverse=False): bins go LOW → HIGH
      e.g. active_day_ct: <5→0, 5-15→5, 15-20→7.5, >20→10

    Inverse (inverse=True) : bins go HIGH → LOW
      e.g. CV_sim: <0.02→10, 0.02-0.05→7.5, ..., >0.2→-5

    Each bin is (upper_bound, score).
    None as upper_bound = catch-all for remaining values.
    """
    if pd.isna(value):
        return np.nan

    for upper_bound, score in bins:
        if upper_bound is None:
            return score
        if value <= upper_bound:
            return score

    # Should never reach here if last bin has None
    return np.nan


def score_row(row, feature_config):
    """
    Score one row across all features.
    Returns dict of individual scores + weighted sum + final score.
    """
    total_weight  = sum(f["weight"] for f in feature_config)
    weighted_sum  = 0.0
    detail        = {}

    for feat in feature_config:
        col     = feat["col"]
        weight  = feat["weight"]
        inverse = feat["inverse"]
        bins    = feat["bins"]

        raw_val = row.get(col, np.nan)
        score   = apply_bins(raw_val, bins, inverse)

        detail[f"score_{col}"] = score

        if not pd.isna(score):
            weighted_sum += score * weight

    final_score = weighted_sum / total_weight if total_weight > 0 else np.nan
    detail["final_score"]    = round(final_score, 2)
    detail["total_weight"]   = total_weight

    return detail


def assign_label(score):
    """
    Assign final label based on score thresholds.
    Only BOT and SUSPECT — no PROSPECT.
    """
    if pd.isna(score):
        return "UNKNOWN"
    if score >= BOT_THRESHOLD:
        return "BOT"
    return "SUSPECT"
